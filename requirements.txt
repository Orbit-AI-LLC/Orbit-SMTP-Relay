# The relay agent needs one third-party package to run: cryptography, for the
# AES-256-GCM sealing of every message. On a node it comes from the
# distribution's python3-cryptography package (38 on Debian 12), so a node can
# be reinstalled years from now without reaching PyPI and still start.
#
# Postfix handles TLS and rate limiting; beyond that the agent needs only the
# standard library to move a file from a directory to an HTTP endpoint and
# retry it until it lands.
cryptography>=38
